// R6S Recoil Control Script for Ash
// Generated on: Oct 26, 2023
// Vertical Compensation: 85%
// Horizontal Compensation: 78%
// Humanization Level: 65%
// Randomization Factor: 42%

#include &lt; Macros.h &gt;

Settings {
    sensitivity = 0.12;
    compensation_mode = "adaptive";
}

RecoilPattern ash_g36c {
    // Primary weapon recoil pattern
    vertical_compensation = 85;
    horizontal_compensation = 78;
    
    // Humanization parameters
    humanization_level = 65;
    randomization_factor = 42;
    
    // Compensation curve
    compensation_curve = [
        {shot = 1, x = 0.0, y = -1.2},
        {shot = 2, x = 0.3, y = -0.8},
        {shot = 3, x = -0.2, y = -1.0},
        {shot = 4, x = 0.4, y = -1.1},
        {shot = 5, x = -0.1, y = -0.9}
    ];
    
    // Apply randomization within safe limits
    apply_randomization(min_range = -0.3, max_range = 0.3);
}

Macro compensated_burst {
    // Burst fire with compensation
    loop(count = 5) {
        shoot();
        wait(randomize_time(base = 80ms, variance = 25ms));
        compensate(recoil_pattern = ash_g36c);
    }
}

// Activation macro
on_event(EVENT_MOUSE_WHEEL_DOWN) {
    execute_macro(compensated_burst);
}

        // DOM Elements
        const operatorsContainer = document.getElementById('operators-container');
        const tabs = document.querySelectorAll('.tab');
        const operatorSearch = document.getElementById('operator-search');
        let currentType = 'attackers';

        // Render operators
        function renderOperators(type) {
            operatorsContainer.innerHTML = '';
            const filteredOps = operators[type].filter(op => 
                op.name.toLowerCase().includes(operatorSearch.value.toLowerCase())
            );
            
            filteredOps.forEach((op, index) => {
                const operatorCard = document.createElement('div');
                operatorCard.className = 'operator-card';
                operatorCard.dataset.index = index;
                operatorCard.dataset.type = type;
                
                operatorCard.innerHTML = `
                    <div class="operator-icon" style="background-image: url('${op.icon}')"></div>
                    <div class="operator-name">${op.name}</div>
                `;
                
                operatorCard.addEventListener('click', () => {
                    document.querySelectorAll('.operator-card').forEach(card => {
                        card.classList.remove('active');
                    });
                    operatorCard.classList.add('active');
                    updateScriptForOperator(op);
                });
                
                operatorsContainer.appendChild(operatorCard);
            });
            
            // Set first operator as active by default
            if (filteredOps.length > 0) {
                const firstCard = operatorsContainer.querySelector('.operator-card');
                firstCard.classList.add('active');
                updateScriptForOperator(filteredOps[0]);
            }
        }

        // Update script for selected operator
        function updateScriptForOperator(operator) {
            const scriptOutput = document.getElementById('script-output');
            const now = new Date();
            const dateStr = now.toLocaleDateString('en-US', { 
                month: 'short', 
                day: 'numeric', 
                year: 'numeric' 
            });
            
            const weaponLower = operator.weapon.toLowerCase().replace(/\s+/g, '');
            
            scriptOutput.textContent = `// R6S Recoil Control Script for ${operator.name}
// Generated on: ${dateStr}
// Vertical Compensation: ${document.getElementById('vertical-slider').value}%
// Horizontal Compensation: ${document.getElementById('horizontal-slider').value}%
// Humanization Level: ${document.getElementById('humanization-slider').value}%
// Randomization Factor: ${document.getElementById('randomization-slider').value}%

#include < Macros.h >

Settings {
    sensitivity = 0.12;
    compensation_mode = "adaptive";
}

RecoilPattern ${operator.name.toLowerCase()}_${weaponLower} {
    // Primary weapon recoil pattern
    vertical_compensation = ${document.getElementById('vertical-slider').value};
    horizontal_compensation = ${document.getElementById('horizontal-slider').value};
    
    // Humanization parameters
    humanization_level = ${document.getElementById('humanization-slider').value};
    randomization_factor = ${document.getElementById('randomization-slider').value};
    
    // Compensation curve
    compensation_curve = [
        {shot = 1, x = 0.0, y = -1.2},
        {shot = 2, x = 0.3, y = -0.8},
        {shot = 3, x = -0.2, y = -1.0},
        {shot = 4, x = 0.4, y = -1.1},
        {shot = 5, x = -0.1, y = -0.9}
    ];
    
    // Apply randomization within safe limits
    apply_randomization(min_range = -0.3, max_range = 0.3);
}

Macro compensated_burst {
    // Burst fire with compensation
    loop(count = 5) {
        shoot();
        wait(randomize_time(base = 80ms, variance = 25ms));
        compensate(recoil_pattern = ${operator.name.toLowerCase()}_${weaponLower});
    }
}

// Activation macro
on_event(EVENT_MOUSE_WHEEL_DOWN) {
    execute_macro(compensated_burst);
}`;
        }

        // Update slider value displays
        const sliders = document.querySelectorAll('input[type="range"]');
        sliders.forEach(slider => {
            const valueDisplay = document.getElementById(`${slider.id.split('-')[0]}-value`);
            
            slider.addEventListener('input', () => {
                valueDisplay.textContent = `${slider.value}%`;
                
                // Update script when sliders change
                const activeCard = document.querySelector('.operator-card.active');
                if (activeCard) {
                    const type = activeCard.dataset.type;
                    const index = activeCard.dataset.index;
                    const operator = operators[type][index];
                    updateScriptForOperator(operator);
                }
                
                // Add animation effect when changing
                valueDisplay.style.transform = 'scale(1.2)';
                setTimeout(() => {
                    valueDisplay.style.transform = 'scale(1)';
                }, 200);
            });
        });

        // Generate button functionality
        document.getElementById('generate-btn').addEventListener('click', () => {
            showNotification();
            simulateProcessing();
        });

        function showNotification() {
            const notification = document.getElementById('notification');
            notification.classList.add('show');
            
            setTimeout(() => {
                notification.classList.remove('show');
            }, 3000);
        }

        function simulateProcessing() {
            const statusLight = document.getElementById('status-light');
            const statusText = document.querySelector('.status-indicator span');
            
            // Processing state
            statusLight.classList.remove('active');
            statusText.textContent = 'Processing...';
            
            setTimeout(() => {
                // Completed state
                statusLight.classList.add('active');
                statusText.textContent = 'Ready';
            }, 1500);
        }

        // Initialize
        renderOperators(currentType);
